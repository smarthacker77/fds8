import pandas as pd

data = {
    'Name': ['Amit', 'Rahul', 'Priya', 'Sneha', 'Amit'],
    'Attendance': [85, 90, 78, None, 85],
    'Score': [80, None, 75, 90, 80]
}

df = pd.DataFrame(data)

print("Original DataFrame:")
print(df)

mean_score = df['Score'].mean()
df['Score'] = df['Score'].fillna(mean_score)

print("\nDataFrame after filling missing scores:")
print(df)

print("\nDuplicate Records:")
print(df.duplicated())

print("\nDuplicate Rows:")
print(df[df.duplicated()])

import numpy as np
import matplotlib.pyplot as plt

ages = np.array(
    
    [22, 25, 28, 30, 32,
    35, 38, 40, 42, 45,
    27, 31, 36, 48, 50]

)

print("Employee Ages:")
print(ages)

bins = [20, 30, 40, 50, 60]

frequency, bin_edges = np.histogram(ages, bins=bins)

print("\nAge Group Frequency:")
for i in range(len(frequency)):
    print(f"{bin_edges[i]}-{bin_edges[i+1]}: {frequency[i]}")

plt.hist(ages, bins=bins, color='skyblue', edgecolor='black')

plt.title("Histogram of Employee Ages")
plt.xlabel("Age Groups")
plt.ylabel("Frequency")
plt.xticks(bins)
plt.grid(axis='y', linestyle='--', alpha=0.7)

plt.show()
