import numpy as np

marks = np.array([85, 72, 45, 90, 35, 60, 55, 28, 78, 40])

low = medium = high = 0

for m in marks:
    if m < 40:
        low += 1
    elif m < 70:
        medium += 1
    else:
        high += 1

print("Below 40:", low)
print("40 to 69:", medium)
print("70 and above:", high)

import numpy as np
import matplotlib.pyplot as plt

d1 = np.random.normal(50, 10, 1000)
d2 = np.random.normal(70, 15, 1000)

plt.hist(d1, bins=20, alpha=0.6, edgecolor='black', label='Dataset 1')
plt.hist(d2, bins=20, alpha=0.6, edgecolor='black', label='Dataset 2')

plt.xlabel("Values")
plt.ylabel("Frequency")
plt.title("Normal Distributions")
plt.legend()
plt.show()
